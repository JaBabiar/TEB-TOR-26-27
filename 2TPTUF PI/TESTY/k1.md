# Kartkówka - Obsługa baz danych w PHP

## Cz1 - Struktura bazy

```sql
CREATE DATABASE IF NOT EXISTS kartkowka_szkola;
USE kartkowka_szkola;


CREATE TABLE uczniowie ( 
    id INT AUTO_INCREMENT PRIMARY KEY,
    imie varchar(50),
    nazwisko varchar(50),
    klasa varchar(50)
);

INSERT INTO uczniowie(imie, nazwisko, klasa) VALUES
('Jan', 'Kowalski', '2TP'),
('Adam', 'Abacki', '2TP'),
('Bartosz', 'Babacki', '3TP');

```
